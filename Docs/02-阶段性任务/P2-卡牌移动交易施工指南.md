void UBattleGameFlowSubsystem::EnqueueTransaction(UBattleTransaction* Tx) {
    if (!IsRunningOnServer()) return;          // 服务器门控
    if (!Tx) return;
    PendingTransactions.Add(Tx);               // ✔ 缺陷②消除：Tx 真正入队
    if (ActiveTransaction.IsValid()) return;   // ✔ 忙锁排队：Execute 期新 Enqueue 只入队、不递归驱动

}
void UBattleGameFlowSubsystem::DriveTransaction() {
    if (!IsRunningOnServer()) return;
    if (PendingTransactions.IsEmpty()) return; // ✔ 缺陷③消除：空队列守卫，首发不再越界
    ActiveTransaction = PendingTransactions[0];
    PendingTransactions.RemoveAt(0);
    ActiveTransaction->Execute();              // Execute 现为基类 public，驱动直调可行（09-09）
}
void UBattleGameFlowSubsystem::OnTransactionFinished(UBattleTransaction* Tx) {
    if (!IsRunningOnServer()) return;
    if (ActiveTransaction != Tx) return;       // 收尾交易须为 Active，防串扰
    ActiveTransaction = nullptr;
    DriveTransaction();                        // ✔ 收尾踢链：自然驱动下一条
}

## 8. 设计备忘（09-09）：FArkCard 不承载归属（Owner）

- **议题**：是否给 `FArkCard` 增加 `Owner` 成员——进入玩家任意区域置当前玩家、进入弃牌区置
  `nullptr`。
- **语义确认**："牌归谁 ≠ 牌在什么容器"成立——牌被使用后（落桌面 `Used` 缓冲堆）至真弃置
  前，仍应视为打出它的玩家的（`08` §7.6：`Used` 不算重置位，入 `Discard` 才终结）。
- **结论（09-09 拍板）：引擎与数据结构不改**——`FArkCard` 不加 `Owner` 成员、区域条目不包
  `{Card, Owner}`；需要该语义的**技能自行实现记录**。
  - 归属在玩家区 / 弃牌堆可由"所在容器宿主"推导（玩家区 = 持有者，`GetContainerActor`；
    弃牌堆无主 = 桌面容器），引擎已有单点（`08` §4）；
  - `FArkCard` 为值类型、区条目 / 交易载荷多拷贝，冗余归属位会引入陈旧副本风险，且易主 /
    入弃的写入点会散落各效果；
  - "Used 期仍归打出者"属技能语义需求（如复活 / 回手还给出牌者、使用痕迹只随入弃清理）——
    由需要它的技能挂实例标记 / 来源自记录（`08` §7.6 终结类别、`06` 来源链），引擎只保留
    "入弃即终"的剥离位，不做全局归属特判。
- **落点**：本备忘不产生 P2 行动项（§E/F/G 不变）；若未来出现"所有者 ≠ 当前持有者"双轴
  （借调 / 驻场归原主）语义，再单独评审建模。
